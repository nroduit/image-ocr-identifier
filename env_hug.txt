# Environment for the container image (docker/Dockerfile, CPU build).
# Usage: docker run --env-file .env.docker -p 8000:8000 <image>
#
# IMPORTANT: every path here refers to a location INSIDE the container, not the
# host. Do not point MODEL_DATA_PATH at a host directory: --env-file overrides
# the image defaults, and the models are baked in at /app/data/models.

OCR_DEVICE=gpu

# --- Models (baked into the image by the Dockerfile) ---
MODEL_DATA_PATH=/app/data
OCR_DET_MODEL=PP-OCRv6_medium_det
